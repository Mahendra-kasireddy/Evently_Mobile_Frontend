import { useEffect, useRef, useState } from 'react';
import {
  Animated,
  Easing,
  Modal,
  Platform,
  Pressable,
  StyleSheet,
  View,
  type LayoutChangeEvent,
} from 'react-native';
import type { BottomTabBarProps } from '@react-navigation/bottom-tabs';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import Svg, { Path } from 'react-native-svg';
import { GooeyBubbles } from './GooeyBubbles';
import {
  EventlyIcon,
  EventlyText,
  GradientFill,
  useReducedMotion,
} from '../Components';

/** Each tab's look: its gradient, and its icon filled and outlined. */
const TAB_LOOK: Record<
  string,
  {
    gradient: [string, string];
    ink: string;
    on: string;
    off: string;
    label: string;
  }
> = {
  Home: {
    gradient: ['#ff8a5c', '#e8433a'],
    ink: '#e2552f',
    on: 'home-variant',
    off: 'home-variant-outline',
    label: 'Home',
  },
  BookingsTab: {
    gradient: ['#a084ff', '#5a35e0'],
    ink: '#6d4df2',
    on: 'calendar-check',
    off: 'calendar-check-outline',
    label: 'Bookings',
  },
  Plan: {
    gradient: ['#a084ff', '#5a35e0'],
    ink: '#6d4df2',
    on: 'clipboard-text',
    off: 'clipboard-text-outline',
    label: 'Plan',
  },
  Events: {
    gradient: ['#3cc9a1', '#0e8a68'],
    ink: '#0f8a68',
    on: 'ticket-confirmation',
    off: 'ticket-confirmation-outline',
    label: 'Events',
  },
  Chat: {
    gradient: ['#5b9bff', '#2554b8'],
    ink: '#2b5aa8',
    on: 'chat-processing',
    off: 'chat-processing-outline',
    label: 'Chat',
  },
};

const FALLBACK = TAB_LOOK.Home;
/** Tab screens with no button on the bar — reached another way (Plan: the + menu). */
const OFF_BAR = new Set(['Plan']);
const MUTED = '#8a93a3';

/* The bar and the raised button in its notch. */
const BAR_HEIGHT = 66;
const BAR_RADIUS = 24;
const FAB_SIZE = 60;
/** The notch's radius: the button plus a ring of air around it. */
const NOTCH_R = FAB_SIZE / 2 + 7;
/** How far the button rises above the bar's top edge. */
const FAB_RISE = 22;
const FAB_GRADIENT: [string, string] = ['#ff7eb3', '#e8433a'];

/** What the + offers, and where each bubble lands relative to the button. */
type QuickAction = {
  key: 'plan' | 'guests' | 'chat';
  label: string;
  icon: string;
  dx: number;
  dy: number;
};
const QUICK_ACTIONS: QuickAction[] = [
  { key: 'plan', label: 'New plan', icon: 'calendar-plus', dx: -100, dy: -78 },
  {
    key: 'guests',
    label: 'Add guests',
    icon: 'account-multiple-plus-outline',
    dx: 0,
    dy: -122,
  },
  { key: 'chat', label: 'Chat', icon: 'chat-plus-outline', dx: 100, dy: -78 },
];

/**
 * The bar's outline: a rounded card with a smooth, round notch cut into the
 * middle of its top edge for the button to sit in.
 */
function barPath(w: number): string {
  const h = BAR_HEIGHT;
  const r = BAR_RADIUS;
  const cx = w / 2;
  const R = NOTCH_R;
  return [
    `M ${r} 0`,
    `L ${cx - R - 14} 0`,
    // Shoulder into the notch, the notch itself, and out again.
    `C ${cx - R - 4} 0 ${cx - R} 2 ${cx - R + 1} 8`,
    `A ${R} ${R} 0 0 0 ${cx + R - 1} 8`,
    `C ${cx + R} 2 ${cx + R + 4} 0 ${cx + R + 14} 0`,
    `L ${w - r} 0`,
    `A ${r} ${r} 0 0 1 ${w} ${r}`,
    `L ${w} ${h - r}`,
    `A ${r} ${r} 0 0 1 ${w - r} ${h}`,
    `L ${r} ${h}`,
    `A ${r} ${r} 0 0 1 0 ${h - r}`,
    `L 0 ${r}`,
    `A ${r} ${r} 0 0 1 ${r} 0`,
    'Z',
  ].join(' ');
}

function TabButton({
  routeName,
  label,
  focused,
  onPress,
  onLongPress,
}: {
  routeName: string;
  label: string;
  focused: boolean;
  onPress: () => void;
  onLongPress: () => void;
}) {
  const look = TAB_LOOK[routeName] ?? FALLBACK;
  /* The gradient pill springs in on the tab that becomes current. */
  const grow = useRef(new Animated.Value(focused ? 1 : 0)).current;

  useEffect(() => {
    Animated.spring(grow, {
      toValue: focused ? 1 : 0,
      friction: 7,
      tension: 90,
      useNativeDriver: true,
    }).start();
  }, [focused, grow]);

  return (
    <Pressable
      style={s.tab}
      onPress={onPress}
      onLongPress={onLongPress}
      accessibilityRole="tab"
      accessibilityState={{ selected: focused }}
      accessibilityLabel={label}
      testID={`tab-${routeName}`}
    >
      <View style={s.iconSlot}>
        <Animated.View
          style={[
            s.pill,
            {
              opacity: grow,
              transform: [
                {
                  scale: grow.interpolate({
                    inputRange: [0, 1],
                    outputRange: [0.6, 1],
                  }),
                },
              ],
            },
          ]}
          pointerEvents="none"
        >
          <GradientFill colors={look.gradient} direction="diagonal" />
        </Animated.View>
        <EventlyIcon
          name={focused ? look.on : look.off}
          size={22}
          color={focused ? '#ffffff' : MUTED}
        />
      </View>
      <EventlyText
        variant="caption"
        style={[s.label, focused && s.labelOn, focused && { color: look.ink }]}
        numberOfLines={1}
      >
        {label}
      </EventlyText>
    </Pressable>
  );
}

/**
 * The round button itself: a gradient + that turns a quarter-and-a-bit into
 * an ×, its colour fading out, as the menu opens. Drawn twice — in the bar,
 * and inside the open menu exactly over it — so the menu's backdrop never
 * dims it.
 */
function FabFace({ progress }: { progress: Animated.Value }) {
  return (
    <>
      {/* The colour fades out as it opens: the × is a white mark in a white
          ring, with nothing filled behind it. */}
      <Animated.View
        style={[
          s.fabFill,
          {
            opacity: progress.interpolate({
              inputRange: [0, 1],
              outputRange: [1, 0],
              extrapolate: 'clamp',
            }),
          },
        ]}
      >
        <GradientFill colors={FAB_GRADIENT} direction="diagonal" />
      </Animated.View>
      <Animated.View
        style={{
          transform: [
            {
              rotate: progress.interpolate({
                inputRange: [0, 1],
                outputRange: ['0deg', '135deg'],
              }),
            },
          ],
        }}
      >
        <EventlyIcon name="plus" size={30} color="#ffffff" />
      </Animated.View>
    </>
  );
}

/**
 * The app's bottom bar: Home and Bookings, a raised + in a notch, Events and
 * Chat.
 *
 * The + is the shortcut to starting something — a new plan, guests for an
 * invitation, a chat — so those are one tap from anywhere rather than a tab
 * and a scroll away. It opens three liquid bubbles over a dimmed screen (see
 * GooeyBubbles); tapping outside, the ×, or Android's back closes it.
 *
 * Presses on the tabs follow React Navigation's own contract — emit
 * `tabPress`, respect a prevented default — so screens that listen for a
 * re-tap still hear it.
 */
export function EventlyTabBar({
  state,
  descriptors,
  navigation,
}: BottomTabBarProps) {
  const insets = useSafeAreaInsets();
  const reduced = useReducedMotion();
  const [barWidth, setBarWidth] = useState(0);
  const [open, setOpen] = useState(false);
  /** Where the button sits on screen, so the open menu can draw it in place. */
  const [fabAt, setFabAt] = useState<{ x: number; y: number } | null>(null);
  const fabRef = useRef<View>(null);
  /*
   * The bar's button runs on Animated (native driver). The open menu runs on
   * its own frame loop instead: its liquid bubbles are SVG paths recomputed
   * from one number every frame (see GooeyBubbles), and on this setup an
   * Animated spring reported only its first and last values to listeners —
   * the bubbles never drew. A plain frame loop here does.
   * `menu` mirrors that number for the backdrop and the × in the Modal.
   */
  const progress = useRef(new Animated.Value(0)).current;
  const menu = useRef(new Animated.Value(0)).current;
  const [menuT, setMenuT] = useState(0);
  const loop = useRef<{ x: number; frame: number | null }>({
    x: 0,
    frame: null,
  });

  const setMenu = (x: number) => {
    loop.current.x = x;
    // The dim and the × arrive in the first part of the opening.
    menu.setValue(Math.min(1, Math.max(0, x * 2.5)));
    setMenuT(x);
  };

  /**
   * Open one bubble at a time (see GooeyBubbles' slots); shut in about half
   * the time, last one home first. `t` runs linearly in time and the bubbles
   * do their own easing.
   */
  const runMenu = (to: 0 | 1, done?: () => void) => {
    const st = loop.current;
    if (st.frame !== null) cancelAnimationFrame(st.frame);
    if (reduced) {
      setMenu(to);
      done?.();
      return;
    }
    const from = st.x;
    const duration = (to === 1 ? 750 : 400) * Math.abs(to - from);
    const started = Date.now();
    const step = () => {
      const k =
        duration > 0 ? Math.min(1, (Date.now() - started) / duration) : 1;
      setMenu(from + (to - from) * k);
      if (k >= 1) {
        st.frame = null;
        done?.();
      } else {
        st.frame = requestAnimationFrame(step);
      }
    };
    st.frame = requestAnimationFrame(step);
  };
  useEffect(() => {
    const st = loop.current;
    return () => {
      if (st.frame !== null) cancelAnimationFrame(st.frame);
    };
  }, []);

  const animateButton = (to: 0 | 1) => {
    if (reduced) {
      progress.setValue(to);
      return;
    }
    (to === 1
      ? Animated.spring(progress, {
          toValue: 1,
          friction: 7,
          tension: 70,
          useNativeDriver: true,
        })
      : Animated.timing(progress, {
          toValue: 0,
          duration: 220,
          easing: Easing.in(Easing.cubic),
          useNativeDriver: true,
        })
    ).start();
  };

  const openMenu = () => {
    fabRef.current?.measureInWindow((x, y, w, h) => {
      setFabAt({ x: x + w / 2, y: y + h / 2 });
      setMenu(0);
      setOpen(true);
      animateButton(1);
    });
  };
  /** The bubbles start once the Modal is actually on screen. */
  const onMenuShown = () => runMenu(1);
  const closeMenu = (then?: () => void) => {
    animateButton(0);
    runMenu(0, () => {
      setOpen(false);
      then?.();
    });
  };

  const runAction = (key: QuickAction['key']) => {
    closeMenu(() => {
      // Root-stack and tab routes alike: navigate bubbles up to whichever
      // navigator owns the name.
      const nav = navigation as unknown as {
        navigate: (name: string, params?: object) => void;
      };
      if (key === 'plan') nav.navigate('Plan');
      else if (key === 'guests') nav.navigate('GuestList');
      else nav.navigate('Chat');
    });
  };

  const tabs = state.routes.flatMap((route, index) => {
    if (OFF_BAR.has(route.name)) return [];
    const focused = state.index === index;
    const options = descriptors[route.key]?.options ?? {};
    const label =
      typeof options.tabBarLabel === 'string'
        ? options.tabBarLabel
        : TAB_LOOK[route.name]?.label ?? route.name;

    const onPress = () => {
      const event = navigation.emit({
        type: 'tabPress',
        target: route.key,
        canPreventDefault: true,
      });
      if (!focused && !event.defaultPrevented) {
        navigation.navigate(route.name, route.params);
      }
    };
    const onLongPress = () =>
      navigation.emit({ type: 'tabLongPress', target: route.key });

    return [
      <TabButton
        key={route.key}
        routeName={route.name}
        label={label}
        focused={focused}
        onPress={onPress}
        onLongPress={onLongPress}
      />,
    ];
  });
  const half = Math.ceil(tabs.length / 2);

  return (
    <View style={[s.wrap, { paddingBottom: Math.max(insets.bottom, 10) }]}>
      <View
        style={s.barBox}
        onLayout={(e: LayoutChangeEvent) =>
          setBarWidth(e.nativeEvent.layout.width)
        }
      >
        {barWidth > 0 ? (
          <View style={s.barShadow} pointerEvents="none">
            <Svg width={barWidth} height={BAR_HEIGHT}>
              <Path
                d={barPath(barWidth)}
                fill="#ffffff"
                stroke="#ece8f3"
                strokeWidth={Platform.OS === 'android' ? 1 : 0}
              />
            </Svg>
          </View>
        ) : null}
        <View style={s.row}>
          {tabs.slice(0, half)}
          <View style={s.notchGap} />
          {tabs.slice(half)}
        </View>
      </View>

      {/* A child of the strip, not of the bar: Android delivers no touches
          outside a parent's box, and the button rises above the bar's. */}
      <Pressable
        ref={fabRef}
        /* Hidden while the menu is open — its copy in the Modal, with no
           fill, sits exactly here, and this one would show through it. */
        style={[s.fab, open && s.fabHidden]}
        onPress={openMenu}
        accessibilityRole="button"
        accessibilityLabel="Create"
        accessibilityHint="Start a plan, add guests or open a chat"
        accessibilityState={{ expanded: open }}
        testID="tab-create"
      >
        <FabFace progress={progress} />
      </Pressable>

      <Modal
        visible={open}
        transparent
        animationType="none"
        statusBarTranslucent
        onShow={onMenuShown}
        onRequestClose={() => closeMenu()}
      >
        <Pressable
          style={StyleSheet.absoluteFill}
          onPress={() => closeMenu()}
          accessibilityLabel="Close"
          testID="create-backdrop"
        >
          <Animated.View
            style={[StyleSheet.absoluteFill, s.backdrop, { opacity: menu }]}
          />
        </Pressable>

        {fabAt ? (
          <>
            <GooeyBubbles
              t={menuT}
              fab={fabAt}
              actions={QUICK_ACTIONS}
              colors={FAB_GRADIENT}
              onPick={key => runAction(key as QuickAction['key'])}
            />

            {/* The same button, over the backdrop, now an ×. */}
            <Pressable
              style={[
                s.fab,
                s.fabFloating,
                s.fabOpen,
                { left: fabAt.x - FAB_SIZE / 2, top: fabAt.y - FAB_SIZE / 2 },
              ]}
              onPress={() => closeMenu()}
              accessibilityRole="button"
              accessibilityLabel="Close"
              testID="create-close"
            >
              <FabFace progress={menu} />
            </Pressable>
          </>
        ) : null}
      </Modal>
    </View>
  );
}

const s = StyleSheet.create({
  /* The strip under the card matches the app's canvas, so the card floats. */
  wrap: {
    backgroundColor: '#faf8f7',
    paddingHorizontal: 14,
    paddingTop: FAB_RISE + 4,
  },
  barBox: { height: BAR_HEIGHT },
  barShadow: {
    position: 'absolute',
    top: 0,
    left: 0,
    right: 0,
    bottom: 0,
    ...Platform.select({
      ios: {
        shadowColor: '#1a2e5a',
        shadowOpacity: 0.1,
        shadowRadius: 16,
        shadowOffset: { width: 0, height: 6 },
      },
      android: {},
    }),
  },
  row: {
    flexDirection: 'row',
    alignItems: 'center',
    height: BAR_HEIGHT,
    paddingHorizontal: 6,
  },
  notchGap: { width: NOTCH_R * 2 + 4 },
  tab: { flex: 1, alignItems: 'center', justifyContent: 'center', gap: 3 },
  iconSlot: {
    width: 52,
    height: 32,
    alignItems: 'center',
    justifyContent: 'center',
  },
  pill: {
    position: 'absolute',
    top: 0,
    left: 0,
    right: 0,
    bottom: 0,
    borderRadius: 16,
    overflow: 'hidden',
  },
  label: { color: MUTED, fontSize: 11, lineHeight: 14, fontWeight: '500' },
  labelOn: { fontWeight: '700' },

  fab: {
    position: 'absolute',
    top: 2,
    alignSelf: 'center',
    width: FAB_SIZE,
    height: FAB_SIZE,
    borderRadius: FAB_SIZE / 2,
    borderWidth: 3,
    borderColor: '#ffffff',
    alignItems: 'center',
    justifyContent: 'center',
    backgroundColor: '#e8433a',
    ...Platform.select({
      ios: {
        shadowColor: '#e2477a',
        shadowOpacity: 0.4,
        shadowRadius: 12,
        shadowOffset: { width: 0, height: 6 },
      },
      android: { elevation: 8 },
    }),
  },
  fabFloating: { alignSelf: 'auto' },
  fabHidden: { opacity: 0 },
  /* No fill and no shadow behind the ×, only the ring. */
  fabOpen: {
    backgroundColor: 'transparent',
    elevation: 0,
    shadowOpacity: 0,
  },
  fabFill: {
    position: 'absolute',
    top: 0,
    left: 0,
    right: 0,
    bottom: 0,
    borderRadius: FAB_SIZE / 2,
    overflow: 'hidden',
  },

  backdrop: { backgroundColor: 'rgba(20, 14, 40, 0.72)' },
});

export default EventlyTabBar;
